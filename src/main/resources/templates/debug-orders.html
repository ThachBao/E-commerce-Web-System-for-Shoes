📝 SQL để kiểm tra database

-- Xem tất cả đơn hàng
SELECT * FROM Orders ORDER BY id DESC;

-- Xem đơn hàng với thông tin user
SELECT o.*, u.fullName 
FROM Orders o 
LEFT JOIN Users u ON o.userId = u.id 
ORDER BY o.id DESC;

-- Xem order items
SELECT oi.*, o.orderCode 
FROM Order_Item oi 
JOIN Orders o ON oi.orderId = o.id 
ORDER BY oi.id DESC;

-- Xem payment
SELECT p.*, o.orderCode 
FROM Payment p 
JOIN Orders o ON p.orderId = o.id 
ORDER BY p.id DESC;
